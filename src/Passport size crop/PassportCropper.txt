import java.awt.*;
import java.awt.event.*;
import java.io.*;
import javax.swing.*;
import javax.swing.filechooser.FileNameExtensionFilter;
import com.formdev.flatlaf.FlatLightLaf;
import org.opencv.core.*;
import org.opencv.imgcodecs.Imgcodecs;
import org.opencv.imgproc.Imgproc;
import org.opencv.objdetect.CascadeClassifier;

public class FaceCropper extends JFrame {
    private JTextField inputField, outputField;
    private JProgressBar progressBar;
    private JButton startButton;
    private JLabel statusLabel;
    private File inputDir, outputDir;
    private volatile boolean isProcessing = false;

    static {
        // Load OpenCV native library
        System.load("C:\\FaceCropper\\lib\\opencv_java455.dll");
    }

    public FaceCropper() {
        // Set FlatLaf look and feel
        try {
            UIManager.setLookAndFeel(new FlatLightLaf());
        } catch (Exception e) {
            e.printStackTrace();
        }

        setTitle("Face Cropper");
        setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        setSize(600, 400);
        setLocationRelativeTo(null);
        setLayout(new GridBagLayout());
        GridBagConstraints gbc = new GridBagConstraints();
        gbc.insets = new Insets(10, 10, 10, 10);
        gbc.fill = GridBagConstraints.HORIZONTAL;

        // Input directory selection
        JLabel inputLabel = new JLabel("Input Directory:");
        gbc.gridx = 0;
        gbc.gridy = 0;
        add(inputLabel, gbc);

        inputField = new JTextField(30);
        inputField.setEditable(false);
        gbc.gridx = 1;
        gbc.gridy = 0;
        add(inputField, gbc);

        JButton inputButton = new JButton("Browse...");
        gbc.gridx = 2;
        gbc.gridy = 0;
        add(inputButton, gbc);
        inputButton.addActionListener(e -> chooseDirectory(true));

        // Output directory selection
        JLabel outputLabel = new JLabel("Output Directory:");
        gbc.gridx = 0;
        gbc.gridy = 1;
        add(outputLabel, gbc);

        outputField = new JTextField(30);
        outputField.setEditable(false);
        gbc.gridx = 1;
        gbc.gridy = 1;
        add(outputField, gbc);

        JButton outputButton = new JButton("Browse...");
        gbc.gridx = 2;
        gbc.gridy = 1;
        add(outputButton, gbc);
        outputButton.addActionListener(e -> chooseDirectory(false));

        // Start button
        startButton = new JButton("Start Cropping");
        gbc.gridx = 0;
        gbc.gridy = 2;
        gbc.gridwidth = 3;
        add(startButton, gbc);
        startButton.addActionListener(e -> startCropping());

        // Progress bar
        progressBar = new JProgressBar(0, 100);
        progressBar.setStringPainted(true);
        gbc.gridx = 0;
        gbc.gridy = 3;
        gbc.gridwidth = 3;
        add(progressBar, gbc);

        // Status label
        statusLabel = new JLabel("Ready");
        gbc.gridx = 0;
        gbc.gridy = 4;
        gbc.gridwidth = 3;
        add(statusLabel, gbc);
    }

    private void chooseDirectory(boolean isInput) {
        JFileChooser chooser = new JFileChooser();
        chooser.setFileSelectionMode(JFileChooser.DIRECTORIES_ONLY);
        if (chooser.showOpenDialog(this) == JFileChooser.APPROVE_OPTION) {
            File selectedDir = chooser.getSelectedFile();
            if (isInput) {
                inputDir = selectedDir;
                inputField.setText(inputDir.getAbsolutePath());
            } else {
                outputDir = selectedDir;
                outputField.setText(outputDir.getAbsolutePath());
            }
        }
    }

    private void startCropping() {
        if (isProcessing) {
            statusLabel.setText("Processing already in progress!");
            return;
        }
        if (inputDir == null || !inputDir.isDirectory()) {
            statusLabel.setText("Please select a valid input directory.");
            return;
        }
        if (outputDir == null || !outputDir.exists()) {
            statusLabel.setText("Please select a valid output directory.");
            return;
        }

        isProcessing = true;
        startButton.setEnabled(false);
        statusLabel.setText("Processing...");
        progressBar.setValue(0);

        new Thread(() -> {
            try {
                processImages();
                SwingUtilities.invokeLater(() -> {
                    statusLabel.setText("Processing completed!");
                    progressBar.setValue(100);
                    startButton.setEnabled(true);
                    isProcessing = false;
                });
            } catch (Exception ex) {
                SwingUtilities.invokeLater(() -> {
                    statusLabel.setText("Error: " + ex.getMessage());
                    startButton.setEnabled(true);
                    isProcessing = false;
                });
            }
        }).start();
    }

    private void processImages() {
        // Initialize OpenCV
        System.loadLibrary(Core.NATIVE_LIBRARY_NAME);
        CascadeClassifier faceDetector = new CascadeClassifier("C:\\FaceCropper\\models\\haarcascade_frontalface_default.xml");
        if (faceDetector.empty()) {
            throw new RuntimeException("Failed to load Haar cascade classifier.");
        }

        File[] imageFiles = inputDir.listFiles((dir, name) ->
                name.toLowerCase().endsWith(".jpg") || name.toLowerCase().endsWith(".png") ||
                name.toLowerCase().endsWith(".jpeg"));

        if (imageFiles == null || imageFiles.length == 0) {
            throw new RuntimeException("No images found in input directory.");
        }

        int total = imageFiles.length;
        int processed = 0;

        for (File imageFile : imageFiles) {
            Mat image = Imgcodecs.imread(imageFile.getAbsolutePath());
            if (image.empty()) {
                continue;
            }

            // Convert to grayscale for face detection
            Mat gray = new Mat();
            Imgproc.cvtColor(image, gray, Imgproc.COLOR_BGR2GRAY);

            // Detect faces
            MatOfRect faces = new MatOfRect();
            faceDetector.detectMultiScale(gray, faces, 1.1, 3, 0, new Size(30, 30), new Size());

            // Crop the largest face
            Rect[] faceArray = faces.toArray();
            if (faceArray.length > 0) {
                Rect largestFace = faceArray[0];
                for (Rect face : faceArray) {
                    if (face.width * face.height > largestFace.width * largestFace.height) {
                        largestFace = face;
                    }
                }

                // Add padding (20% of face size)
                int padding = (int) (largestFace.width * 0.2);
                int x = Math.max(0, largestFace.x - padding);
                int y = Math.max(0, largestFace.y - padding);
                int width = Math.min(image.cols() - x, largestFace.width + 2 * padding);
                int height = Math.min(image.rows() - y, largestFace.height + 2 * padding);

                // Crop face
                Mat cropped = new Mat(image, new Rect(x, y, width, height));

                // Resize to 256x256
                Mat resized = new Mat();
                Imgproc.resize(cropped, resized, new Size(256, 256));

                // Save cropped image
                String outputPath = new File(outputDir, "face_" + imageFile.getName()).getAbsolutePath();
                Imgcodecs.imwrite(outputPath, resized);
            }

            processed++;
            int progress = (int) ((processed / (double) total) * 100);
            SwingUtilities.invokeLater(() -> progressBar.setValue(progress));
        }
    }

    public static void main(String[] args) {
        SwingUtilities.invokeLater(() -> {
            new FaceCropper().setVisible(true);
        });
    }
}
